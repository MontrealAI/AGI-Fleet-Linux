import assert from 'node:assert/strict';
import {readFileSync} from 'node:fs';
import {createHash} from 'node:crypto';
import {canonical as before} from './fleet-contract-before-object-width.mjs';
import {canonical as after} from './alpha-discovery/examples/successor-capability/discovery-lab-contract.mjs';

let seed = 485064;
const random = max => { seed = (Math.imul(seed, 1664525) + 1013904223) >>> 0; return seed % max; };
const scalars = [null, true, false, 0, 1, -1, 9007199254740991, 'é', '😀', '\u0000', '"\\', ''];
function generated(depth = 0) {
    if (depth > 4 || random(3) === 0) return scalars[random(scalars.length)];
    const children = Array.from({length: random(9)}, () => generated(depth + 1));
    return random(2) ? children : Object.fromEntries(children.map((value, index) => ['k' + index, value]));
}
const cases = Array.from({length: 1000}, () => generated());
cases.push(Array(99_999).fill(null), Object.fromEntries(Array.from({length: 99_999}, (_, i) => ['k' + i, 0])),
    'x'.repeat(1_999_998), '\u0000'.repeat(333_333), 'é'.repeat(999_999), '😀'.repeat(499_999) + 'é',
    Object.assign(Object.create(null), {['__proto__']: 0, ['"\\']: 'é'}));
const native = JSON.parse(readFileSync('./fleet-alpha1665-native24/semantic-seam.json', 'utf8'));
for (const record of native.cases ?? []) if (record.request) cases.push(record.request);
for (const file of ['browser-workspace.json', 'manager-review-draft.json', 'jobs-review-draft.json']) {
    cases.push(JSON.parse(readFileSync('./discovery-lab-exact-1665cb83/browser/' + file, 'utf8')));
}
const digest = createHash('sha256');
for (const value of cases) {
    const a = before(value), b = after(value);
    assert.equal(b, a);
    digest.update(a + '\n');
}
console.log(JSON.stringify({runtime: process.version, baselineContractSha256:
    createHash('sha256').update(readFileSync('./fleet-contract-before-object-width.mjs')).digest('hex'),
    newContractSha256: createHash('sha256').update(readFileSync(
        './alpha-discovery/examples/successor-capability/discovery-lab-contract.mjs')).digest('hex'),
    cases: cases.length, allAdmittedCanonicalBytesIdentical: true, aggregateSha256: digest.digest('hex')}, null, 2));
